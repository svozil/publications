\title{Independent Non-Redundant Open Conjectures in Number Theory\\
A Minimal Core of 115 Pairwise Independent Problems}
\author{Karl Svozil}
\affiliation{Independent Compilation}
\date{\today}

\begin{abstract}
We list 115 open conjectures in number theory that are pairwise independent in the sense that no one is known to imply another after collapsing standard implication chains: Grand RH $\Rightarrow$ RH, GRH, ERH, Lindel\"of, Density; Firoozbakht $\Rightarrow$ Cram\'er, Legendre, Oppermann, Andrica, Brocard-gap; Bateman-Horn $\Rightarrow$ Schinzel H $\Rightarrow$ Dickson $\Rightarrow$ Bunyakovsky $\Rightarrow$ $k$-tuple $\Rightarrow$ Twin, Polignac $k$, $n^2+1$; Vojta $\Rightarrow$ abc $\Leftrightarrow$ Szpiro $\Rightarrow$ Pillai, Hall, Fermat-Catalan; Bloch-Kato $\Rightarrow$ BSD $\Rightarrow$ Parity, Sha finiteness; Standard $\Rightarrow$ Hodge, Tate; Sarnak $\Rightarrow$ Chowla; Schanuel $\Rightarrow$ Four Exponentials, $e+\pi$, $e^e$ transcendence. To maximize independent count we drop the 5 very general polynomial-prime conjectures and keep specific prime families, and keep the strongest member of each remaining chain. Weak Goldbach (Helfgott 2013) is excluded as resolved.
\end{abstract}

\section{Implication collapse}
\begin{itemize}
\item RH family kept as Grand RH.
\item Prime gaps family kept as Firoozbakht.
\item Polynomial primes family dropped (Bateman-Horn, Schinzel H, Dickson, Bunyakovsky, $k$-tuple) to keep specific forms independent.
\item Polignac kept, Twin/Cousin/Sexy dropped.
\item abc family kept as Vojta.
\item BSD family kept as Bloch-Kato.
\item Cycles family kept as Standard Conjectures.
\item M\"obius family kept as Sarnak.
\item Transcendence family kept as Schanuel.
\end{itemize}

\section{Independent list}

\begin{enumerate}[label={\bf \arabic*.}, leftmargin=*, itemsep=2pt]

\item \textbf{Grand Riemann Hypothesis:} Every non-trivial zero of any automorphic $L$-function $L(s,\pi)$ has $\Re(s)=1/2$.

\item \textbf{Firoozbakht's Conjecture:} $p_n^{1/n}$ strictly decreasing: $p_{n+1}^{1/(n+1)}<p_n^{1/n}$ for $n\ge1$.

\item \textbf{Fortune's Conjecture:} Let $p_n\#=2\cdot3\cdots p_n$. Let $m>1$ minimal with $p_n\#+m$ prime. Then $m$ (Fortunate number) is prime.

\item \textbf{Gilbreath's Conjecture:} Starting from primes $a_0(n)=p_n$, define $a_{k+1}(n)=|a_k(n+1)-a_k(n)|$. Then $a_k(1)=1$ for all $k\ge1$.

\item \textbf{Grimm's Conjecture:} If $n+1,\dots,n+k$ are composite, there exist distinct primes $p_i$ with $p_i\mid n+i$.

\item \textbf{Artin Primitive Root:} For $a\neq-1$ and not a perfect square, $a$ is primitive root mod $p$ for infinitely many $p$.

\item \textbf{Mersenne infinitude:} Infinitely many primes $M_p=2^p-1$, $p$ prime.

\item \textbf{Fermat finiteness:} Only five Fermat primes $F_n=2^{2^n}+1$ for $n=0,\dots,4$: $3,5,17,257,65537$.

\item \textbf{Repunit prime infinitude:} Infinitely many primes $R_n=(10^n-1)/9$ base $10$.

\item \textbf{Wagstaff prime infinitude:} Infinitely many primes $(2^p+1)/3$, $p$ prime.

\item \textbf{Fibonacci prime infinitude:} Infinitely many $F_n$ prime, $F_1=F_2=1$, $F_{n+2}=F_{n+1}+F_n$.

\item \textbf{Cullen prime infinitude:} Infinitely many primes $C_n=n2^n+1$.

\item \textbf{Woodall prime infinitude:} Infinitely many primes $W_n=n2^n-1$.

\item \textbf{Wilson prime infinitude:} $p^2\mid(p-1)!+1$. Only $5,13,563$ known. Infinitely many?

\item \textbf{Wolstenholme prime infinitude:} $p$ with $\binom{2p-1}{p-1}\equiv1\bmod p^4$. Only $16843,2124679$ known.

\item \textbf{Wall-Sun-Sun existence:} Does prime $p$ exist with $p^2\mid F_{p-(\frac{p}{5})}$ where $F$ Fibonacci and $(\frac{p}{5})$ Legendre symbol?

\item \textbf{Wieferich prime infinitude:} $2^{p-1}\equiv1\bmod p^2$. Infinitely many.

\item \textbf{Palindromic prime infinitude base 10:} Prime with decimal expansion reverse equal itself, infinitely many.

\item \textbf{Emirp infinitude:} Prime $p$ with $\text{reverse}_{10}(p)\neq p$ prime, infinitely many.

\item \textbf{Pierpont prime infinitude:} Primes $2^u3^v+1$ infinitely many.

\item \textbf{Carol/Kynea infinitude:} Carol $(2^n-1)^2-2$ and Kynea $(2^n+1)^2-2$ prime infinitely often.

\item \textbf{Factorial prime infinitude:} $n!\pm1$ prime infinitely often.

\item \textbf{Primorial prime infinitude:} $p_n\#\pm1$ prime infinitely often.

\item \textbf{New Mersenne Conjecture:} For odd prime $p$, conditions $p=2^k\pm1$ or $4^k\pm3$, $2^p-1$ prime, $(2^p+1)/3$ prime are equivalent in stated pairing.

\item \textbf{Sierpinski minimal:} $78557$ is smallest $k$ with $k2^n+1$ composite for all $n$.

\item \textbf{Riesel minimal:} $509203$ is smallest $k$ with $k2^n-1$ composite for all $n$.

\item \textbf{Euclid-Mullin contains all primes:} $a_1=2$, $a_{n+1}=$ smallest prime factor of $1+\prod_{i\le n}a_i$. Does $\{a_n\}$ contain every prime?

\item \textbf{Brocard's Problem:} $n!+1=m^2$ only $n=4,5,7$.

\item \textbf{Goldbach Strong:} Every even $n>2 = p+q$, $p,q$ prime.

\item \textbf{Polignac's Conjecture:} For every even $k$, infinitely many $p$ with $p+k$ prime.

\item \textbf{Sophie Germain infinitude:} Infinitely many $p$ with $2p+1$ prime.

\item \textbf{Landau $n^2+1$:} Infinitely many primes $n^2+1$.

\item \textbf{Prime triplet infinitude:} Infinitely many $(p,p+2,p+6)$ and $(p,p+4,p+6)$.

\item \textbf{Prime quadruplet infinitude:} Infinitely many $(p,p+2,p+6,p+8)$.

\item \textbf{Collatz:} For $n>0$, $n\to n/2$ if even, $3n+1$ if odd, reaches $1$.

\item \textbf{Vojta's Conjecture:} For $X$ smooth, $D$ normal crossings, $h_{K_X+D}(P)\le N^{(1)}(D,P)+d(P)+O(1)$. Implies abc, Szpiro, Pillai, Hall, Fermat-Catalan.

\item \textbf{Beal's Conjecture:} If $A^x+B^y=C^z$ with $x,y,z\ge3$, then $\gcd(A,B,C)>1$.

\item \textbf{Erd\H{o}s-Straus:} For all $n\ge2$, $4/n=1/x+1/y+1/z$ in positive integers.

\item \textbf{Erd\H{o}s-Moser equation:} $1^k+\cdots+(m-1)^k=m^k$ only $1+2=3$.

\item \textbf{Perfect Cuboid:} Does box with integer edges, face diagonals, space diagonal exist?

\item \textbf{Sums of three cubes:} For each $k\not\equiv\pm4\bmod9$, exists $x^3+y^3+z^3=k$.

\item \textbf{Littlewood Conjecture:} $\inf_{n\ge1} n\|n\alpha\|\|n\beta\|=0$ for any real $\alpha,\beta$.

\item \textbf{Bloch-Kato Conjecture:} For motive $M$, $\operatorname{ord}_{s=0}L(M,s)$ equals Selmer rank and special value given by regulator $\times$ Sha etc. Implies BSD.

\item \textbf{Standard Conjectures:} Numerical equivalence = homological equivalence for cycles, Lefschetz standard, Hodge standard. Implies Hodge and Tate.

\item \textbf{Fontaine-Mazur:} Every irreducible odd $p$-adic Galois representation unramified outside finite set and de Rham at $p$ comes from geometry.

\item \textbf{Langlands Reciprocity:} $n$-dimensional Galois representations correspond to automorphic representations of $GL(n)$.

\item \textbf{Artin Holomorphy:} Artin $L$-functions for non-trivial irreducible Galois representations entire.

\item \textbf{Elliott-Halberstam:} $\sum_{q\le x^{1-\epsilon}}\max_{(a,q)=1}|\pi(x;q,a)-\pi(x)/\phi(q)|\ll x/(\log x)^A$.

\item \textbf{Sarnak M\"obius Randomness:} $\sum_{n\le N}\mu(n)f(T^n x)=o(N)$ for zero-entropy $T$. Implies Chowla.

\item \textbf{Sato-Tate general:} For genus $>1$, Frobenius angles equidistributed via Haar measure of Sato-Tate group.

\item \textbf{Montgomery Pair Correlation:} Rescaled zeros of $\zeta$ have pair correlation $1-(\sin\pi u/\pi u)^2$.

\item \textbf{Keating-Snaith:} $\frac1T\int_0^T|\zeta(1/2+it)|^{2k}dt\sim a_k(\log T)^{k^2}$.

\item \textbf{Gauss Circle:} $N(R)=\#\{(x,y)\in\mathbf Z^2:x^2+y^2\le R^2\}=\pi R^2+O(R^{1/4+\epsilon})$.

\item \textbf{Dirichlet Divisor:} $\sum_{n\le x}d(n)=x\log x+(2\gamma-1)x+O(x^{1/4+\epsilon})$.

\item \textbf{Piltz divisor:} $\sum_{n\le x}d_k(n)=xP_{k-1}(\log x)+O(x^{(k-1)/2k+\epsilon})$.

\item \textbf{Waring $G(4)=16$:} Every large integer sum of at most $16$ fourth powers.

\item \textbf{Carmichael Totient:} No $m$ has exactly one $n$ with $\phi(n)=m$.

\item \textbf{Scholz addition chain:} $l(2^n-1)\le n-1+l(n)$ where $l$ minimal addition chain length.

\item \textbf{Leopoldt's Conjecture:} $p$-adic regulator $R_p(K)\neq0$.

\item \textbf{Vandiver's Conjecture:} $p\nmid h^+_p$, $p$ does not divide class number of real subfield of $\mathbf Q(\zeta_p)$.

\item \textbf{Iwasawa $\mu=0$:} $\mu$-invariant of cyclotomic $\mathbf Z_p$-extension is $0$.

\item \textbf{Greenberg's Conjecture:} For totally real field, Iwasawa $\lambda=\mu=0$ for cyclotomic $\mathbf Z_p$-extension.

\item \textbf{Real quadratic class number 1 infinitude:} Infinitely many $d>0$ squarefree with $h(\mathbf Q(\sqrt d))=1$.

\item \textbf{Goormaghtigh:} Only solutions to $(x^m-1)/(x-1)=(y^n-1)/(y-1)$, $m,n>2$, $x\neq y$ give $31$ and $8191$.

\item \textbf{Singmaster's Conjecture:} Multiplicity of $n>1$ in Pascal's triangle bounded by absolute constant, conjectured $\le8$.

\item \textbf{Erd\H{o}s-Tur\'an additive bases:} If $A$ asymptotic basis order $2$, representation function $r_A(n)$ unbounded.

\item \textbf{Erd\H{o}s AP Conjecture:} If $A\subset\mathbf N$ and $\sum_{a\in A}1/a$ diverges, $A$ contains arbitrarily long arithmetic progressions.

\item \textbf{Erd\H{o}s-Selfridge:} $(m+1)\cdots(m+k)$ never perfect power.

\item \textbf{Lonely Runner:} $k$ runners with distinct speeds on unit circle, some time each distance $\ge1/(k+1)$ from origin.

\item \textbf{Frankl Union-Closed:} In finite union-closed family containing non-empty set, some element in at least half sets.

\item \textbf{Davenport constant:} $D(G)$ minimal $n$ such that any sequence of $n$ elements of abelian group $G$ has non-empty zero-sum subsequence. Exact values open.

\item \textbf{EGZ constant:} $s(G)$ minimal $\ell$ such that any sequence of $\ell$ elements has $|G|$ elements summing to zero. Exact values open in higher rank.

\item \textbf{Erd\H{o}s-Fuchs:} If $\sum_{k\le n}r_A(k)=cn+o(n)$ then $o(n)$ not $o(n^{1/4}/\sqrt{\log n})$.

\item \textbf{Lander-Parkin-Selfridge:} If $\sum_{i=1}^m x_i^k=\sum_{j=1}^n y_j^k$ with $m+n<k$, no non-trivial solutions for $k\ge4$.

\item \textbf{Schanuel's Conjecture:} If $z_1,\dots,z_n$ linearly independent over $\mathbf Q$, then $\operatorname{trdeg}_{\mathbf Q}\mathbf Q(z_i,e^{z_i})\ge n$.

\item \textbf{Normality of $\pi$:} Every $k$-digit string appears in base $10$ expansion of $\pi$ with density $10^{-k}$.

\item \textbf{Borel's Conjecture:} Every irrational algebraic number normal in every base.

\item \textbf{Copeland-Erd\H{o}s normal:} $0.23571113\ldots$ formed by concatenating primes normal base $10$.

\item \textbf{Champernowne all bases:} $0.1234567891011\ldots$ normal in every base.

\item \textbf{Powers of 2 digit:} Every large $2^n$ contains digit $7$, and only finitely many $2^n$ avoid digit $0$.

\item \textbf{Prime geometric progression:} Four distinct primes in geometric progression with rational ratio exist.

\item \textbf{Gaussian Moat:} Cannot walk to infinity stepping only on Gaussian primes with bounded step.

\item \textbf{Markov unicity:} Each Markov number $z$ appears as max of exactly one triple $x^2+y^2+z^2=3xyz$.

\item \textbf{Margenstern:} Every large even integer sum of two practical numbers.

\item \textbf{Odd perfect number existence:} Does $n$ with $\sigma(n)=2n$ odd exist?

\item \textbf{Quasiperfect existence:} Does $n$ with $\sigma(n)=2n+1$ exist?

\item \textbf{Odd weird existence:} Does odd abundant non-semiperfect exist?

\item \textbf{Amicable pairs infinitude:} Infinitely many $a\neq b$ with $\sigma(a)=\sigma(b)=a+b$.

\item \textbf{Sociable chains infinitude:} For each $k$, infinitely many aliquot cycles length $k$.

\item \textbf{Catalan-Dickson:} Every aliquot sequence $n,\sigma(n)-n,\dots$ eventually $0$ or periodic.

\item \textbf{Lychrel 196:} 196 never becomes palindrome under reverse-and-add.

\item \textbf{Mahler $3/2$:} Fractional parts $\{(3/2)^n\}$ not dense, $\inf\{(3/2)^n\}>0$.

\item \textbf{Primary pseudoperfect infinitude:} $n$ with $\sum_{p\mid n}1/p+1/n=1$ infinitely many.

\item \textbf{Superperfect infinitude:} $\sigma(\sigma(n))=2n$ infinitely many.

\item \textbf{Giuga numbers infinitude:} Composite squarefree $n$ with $p\mid n\Rightarrow p\mid n/p-1$ infinitely many.

\item \textbf{Lucas-Carmichael infinitude:} Composite squarefree $n$ with $p\mid n\Rightarrow p+1\mid n+1$ infinitely many.

\item \textbf{Primitive non-deficient infinitude:} Minimal abundant numbers w.r.t divisibility infinitely many.

\item \textbf{Primitive weird infinitude:} Minimal weird numbers infinitely many.

\item \textbf{Hilbert Tenth over $\mathbf Q$:} No algorithm decides existence of rational solution to Diophantine equation.

\item \textbf{Unbounded ranks:} Elliptic curves over $\mathbf Q$ have $\sup\operatorname{rank}=\infty$.

\item \textbf{Colliot-Th\'el\`ene:} Brauer-Manin obstruction only obstruction to Hasse principle for rationally connected varieties.

\item \textbf{Uniformity of rational points:} Number rational points on genus $g\ge2$ curve over $\mathbf Q$ bounded by $C(g)$.

\item \textbf{Bombieri-Lang:} Rational points on variety of general type over number field not Zariski dense.

\item \textbf{Cohen-Lenstra heuristics:} Distribution of class groups of quadratic fields follows predicted probabilities.

\item \textbf{Stark Conjectures:} Leading term of Artin $L$ at $s=0$ equals regulator of Stark units.

\item \textbf{Generalized Ramanujan-Petersson:} Hecke eigenvalues of cuspidal automorphic forms on $GL(n)$ have absolute value $1$.

\item \textbf{Irrationality of $\gamma$:} $\gamma=\lim(H_n-\log n)$ irrational.

\item \textbf{Irrationality of $\zeta(2k+1)$:} $\zeta(5),\zeta(7),\dots$ irrational.

\item \textbf{Algebraic independence odd zeta:} $\zeta(3),\zeta(5),\dots$ algebraically independent over $\mathbf Q$.

\item \textbf{Catalan's constant irrational:} $G=\sum_{n\ge0}(-1)^n/(2n+1)^2$ irrational.

\item \textbf{Erd\H{o}s primitive sets:} For primitive $A$ (no element divides another), $\sum_{a\in A}1/(a\log a)\le\sum_{p}1/(p\log p)$.

\item \textbf{Normality of $e$:} $e$ normal base $10$.

\item \textbf{Lemoine's Conjecture:} Every odd $n>5 = p+2q$ with $p,q$ prime.

\item \textbf{Agoh-Giuga:} $n$ prime iff $B_{n-1}\equiv-1\bmod n$ and $\sum_{p\mid n}1/p-\prod_{p\mid n}1/p\in\mathbf N$, $B$ Bernoulli.

\item \textbf{Erd\H{o}s-Ulam:} Does dense set in plane with all mutual distances rational exist?

\end{enumerate}
